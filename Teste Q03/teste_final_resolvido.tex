		\begin{solution}
			\textcolor{white}{.}\\
			\mybox{
				
				Determinar a quantidade de cada reagente (3pt):\\
				$n_{\ce{CH3COOH}} = \dfrac{m}{M} \Leftrightarrow n_{\ce{CH3COOH}} = \dfrac{100}{60,05} = \SI{1,66}{mol}$\\
				$n_{\ce{CH3I}} = \dfrac{m}{M} \Leftrightarrow n_{\ce{CH3I}} = \dfrac{100}{141,94} = \SI{0,705}{\mol}$\\
				
				Concluir que o reagente limitante é o \ce{CH3I} (3pt).\\
				
				Determinar o $n(\ce{CH3COOCH3})_\text{previsto}$ (3pt):\\
				$n(\ce{CH3COOCH3})_\text{previsto} = n(\ce{CH3I}) = \SI{0,705}{\mole}$\\
				
				Determinar o $n(\ce{CH3COOCH3})_\text{obtido}$ (3pt):\\
				$n(\ce{CH3COOCH3})_\text{obtido} = \dfrac{m}{M} \Leftrightarrow n(\ce{CH3COOCH3})_\text{obtido} = \dfrac{50}{74,09} = \SI{0,675}{\mole}$\\
				
				Determinar o rendimento da reação (4pt):\\
				$\eta(\%) = \dfrac{n(\ce{CH3COOCH3})_\text{obtido}}{n(\ce{CH3COOCH3})_\text{previsto}} \times 100 \Leftrightarrow \eta(\%) = \dfrac{0,675}{0,705}\times100 = 95,7\%$\\
				
			}
		\end{solution}
